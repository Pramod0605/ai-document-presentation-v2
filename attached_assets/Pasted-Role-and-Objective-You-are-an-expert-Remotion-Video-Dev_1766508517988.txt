Example 1: Geometrically Correct Right Triangle (SVG - GOLD STANDARD)

const GeometricTriangle = () => {
    const frame = useCurrentFrame();
    const { fps } = useVideoConfig();
    
    // Define triangle dimensions
    const width = 400;
    const height = 300;
    
    // Calculate hypotenuse for validation
    const hypotenuse = Math.sqrt(width ** 2 + height ** 2);
    
    // Define corner points (pixel-perfect)
    const bottomLeft = { x: 0, y: height };
    const bottomRight = { x: width, y: height };
    const topRight = { x: width, y: 0 };
    
    // Animate drawing using strokeDashoffset
    const drawProgress = spring({ frame, fps, config: { damping: 100 } });
    
    const baseLength = width;
    const heightLength = height;
    const hypLength = hypotenuse;
    
    return (
        <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center' }}>
            <svg width={width + 100} height={height + 100} style={{ overflow: 'visible' }}>
                {/* Horizontal leg (base) */}
                <line 
                    x1={bottomLeft.x} 
                    y1={bottomLeft.y}
                    x2={bottomRight.x} 
                    y2={bottomRight.y}
                    stroke="#4ade80"
                    strokeWidth="8"
                    strokeLinecap="round"
                    strokeDasharray={baseLength}
                    strokeDashoffset={baseLength * (1 - drawProgress)}
                />
                
                {/* Vertical leg (height) */}
                <line 
                    x1={bottomRight.x} 
                    y1={bottomRight.y}
                    x2={topRight.x} 
                    y2={topRight.y}
                    stroke="#4ade80"
                    strokeWidth="8"
                    strokeLinecap="round"
                    strokeDasharray={heightLength}
                    strokeDashoffset={heightLength * (1 - drawProgress)}
                />
                
                {/* Hypotenuse (diagonal) */}
                <line 
                    x1={bottomLeft.x} 
                    y1={bottomLeft.y}
                    x2={topRight.x} 
                    y2={topRight.y}
                    stroke="#facc15"
                    strokeWidth="10"
                    strokeLinecap="round"
                    strokeDasharray={hypLength}
                    strokeDashoffset={hypLength * (1 - drawProgress)}
                />
                
                {/* Right angle indicator (perfect 90°) */}
                <path
                    d={`M ${bottomRight.x - 30} ${bottomRight.y} 
                        L ${bottomRight.x - 30} ${bottomRight.y - 30}
                        L ${bottomRight.x} ${bottomRight.y - 30}`}
                    stroke="#60a5fa"
                    strokeWidth="4"
                    fill="none"
                    opacity={drawProgress}
                />
            </svg>
        </AbsoluteFill>
    );
};

Example 2: Smooth Text Animation with Word-by-Word Reveal

const AnimatedTitle = ({ children }: { children: string }) => {
    const frame = useCurrentFrame();
    const { fps } = useVideoConfig();
    
    const words = children.split(' ');
    
    return (
        <h1 style={{
            fontFamily: 'Arial, sans-serif',
            fontSize: 100,
            fontWeight: 'bold',
            textAlign: 'center',
            color: 'white',
        }}>
            {words.map((word, i) => {
                const delay = i * 5; // Stagger each word by 5 frames
                const wordSpring = spring({
                    frame: frame - delay,
                    fps,
                    config: { damping: 200, stiffness: 300 },
                });
                
                const y = interpolate(wordSpring, [0, 1], [50, 0]);
                const opacity = wordSpring;
                
                return (
                    <span
                        key={i}
                        style={{
                            display: 'inline-block',
                            transform: `translateY(${y}px)`,
                            opacity,
                            marginRight: '0.3em',
                        }}
                    >
                        {word}
                    </span>
                );
            })}
        </h1>
    );
};

Common Pitfalls to Avoid

❌ Using positioned divs with transform: rotate() for lines
❌ Forgetting to use extrapolateRight: 'clamp' on interpolations
❌ Not centering AbsoluteFill content with justifyContent/alignItems
❌ Hardcoding frame numbers instead of using spring/interpolate
❌ Creating triangles without verifying corner alignment
❌ Forgetting to export compositionConfig
❌ Missing default export on main component
