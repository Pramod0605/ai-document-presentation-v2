MANDATORY VISUAL ACTIVATION RULE:

   visual_layer = "show" OR visual_layer = "replace"

   - text is being introduced, OR

ADD THIS Tier-1 validator rule
def validate_visuals_are_shown(section):
    if section.get("visual_beats"):
        segments = section["narration"]["segments"]
        if not any(
            seg["display_directives"]["visual_layer"] in ("show", "replace")
            for seg in segments
        ):
            return error("visuals_never_displayed")

def validate_renderer_subject_match(section):

    renderer = section["renderer"]

        return error("biology_must_use_video")

        return error("recap_must_use_video")
