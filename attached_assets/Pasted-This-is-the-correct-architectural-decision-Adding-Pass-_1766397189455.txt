Here are the exact JSON Schema Definitions you need for the Gemini 2.5 Pro API. I have designed these to be plug-and-play for your Python generation_config.

1. Smart Chunker Schema (Pass 0)
Goal: Force the LLM to identify logical boundaries and assign a specific renderer strategy before writing any script.

Python

SMART_CHUNKER_SCHEMA = {
    "type": "OBJECT",
    "properties": {
        "source_topic": {
            "type": "STRING",
            "description": "The main subject matter of the provided text."
        },
        "topics": {
            "type": "ARRAY",
            "description": "List of logical sub-topics extracted from the text.",
            "items": {
                "type": "OBJECT",
                "properties": {
                    "topic_id": { "type": "STRING" },
                    "title": { "type": "STRING" },
                    "concept_type": {
                        "type": "STRING",
                        "enum": ["process", "definition", "example", "formula", "theory", "fact"],
                        "description": "The primary nature of this chunk."
                    },
                    "source_blocks": {
                        "type": "ARRAY",
                        "items": { "type": "INTEGER" },
                        "description": "The IDs of the raw markdown blocks used for this chunk."
                    },
                    "key_terms": {
                        "type": "ARRAY",
                        "items": { "type": "STRING" }
                    },
                    "suggested_renderer": {
                        "type": "STRING",
                        "enum": ["remotion", "manim", "video"],
                        "description": "remotion=text/simple images, manim=math/physics/complex logic, video=biological/real-world processes"
                    }
                },
                "required": ["topic_id", "title", "concept_type", "source_blocks", "suggested_renderer"]
            }
        }
    },
    "required": ["source_topic", "topics"]
}

3. Recap Director Schema (Pass 1b)
Goal: Force exactly 5 Flashcards (Memory) and exactly 5 Video Prompts (Recap) with high word counts.

Python

RECAP_DIRECTOR_SCHEMA = {
    "type": "OBJECT",
    "properties": {
        "sections": {
            "type": "ARRAY",
            "minItems": 2,
            "maxItems": 2,
            "items": {
                "type": "OBJECT",
                "properties": {
                    "section_type": {
                        "type": "STRING",
                        "enum": ["memory", "recap"]
                    },
                    "renderer": {
                        "type": "STRING",
                        "enum": ["remotion", "video"]
                    },
                    "narration": {
                        "type": "OBJECT",
                        "properties": {
                            "full_text": { "type": "STRING" }
                        },
                        "required": ["full_text"]
                    },
                    # MEMORY SECTION SPECIFIC FIELDS
                    "flashcards": {
                        "type": "ARRAY",
                        "description": "Used only for 'memory' section type. Must be exactly 5 cards.",
                        "minItems": 5,
                        "maxItems": 5,
                        "items": {
                            "type": "OBJECT",
                            "properties": {
                                "mnemonic_letter": { "type": "STRING" },
                                "concept": { "type": "STRING" },
                                "definition": { "type": "STRING" }
                            },
                            "required": ["mnemonic_letter", "concept", "definition"]
                        }
                    },
                    # RECAP SECTION SPECIFIC FIELDS
                    "video_prompts": {
                        "type": "ARRAY",
                        "description": "Used only for 'recap' section type. Must be exactly 5 prompts.",
                        "minItems": 5,
                        "maxItems": 5,
                        "items": {
                            "type": "OBJECT",
                            "properties": {
                                "beat_id": { "type": "STRING" },
                                "prompt": {
                                    "type": "STRING",
                                    "description": "Cinematic visual description, >300 words. NO infographics."
                                },
                                "duration_seconds": { "type": "NUMBER" }
                            },
                            "required": ["beat_id", "prompt", "duration_seconds"]
                        }
                    }
                },
                "required": ["section_type", "renderer", "narration"]
            }
        }
    },
    "required": ["sections"]
}
How to use this in Python
When you initialize your Gemini client (assuming standard Google Gen AI SDK), you pass these objects into the generation_config.

Python

# Example for Recap Director
model = genai.GenerativeModel(
    "gemini-2.0-flash-exp", # or your preferred version
    generation_config=genai.GenerationConfig(
        response_mime_type="application/json",
        response_schema=RECAP_DIRECTOR_SCHEMA
    )
)
