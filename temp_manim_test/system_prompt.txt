from manim import *

class MainScene(Scene):
    def construct(self):
        # Segment 1 (10.0s) - Introduction
        title = Text("Trigonometry", font_size=48).to_edge(UP)
        subtitle = Text("The Study of Triangles", font_size=32, color=BLUE).next_to(title, DOWN)
        
        self.play(Write(title), run_time=1.5)
        self.play(FadeIn(subtitle), run_time=1.0)
        self.wait(2.0)
        self.play(Indicate(title), run_time=0.8)
        self.wait(3.0)
        self.play(title.animate.scale(1.05), run_time=0.7)
        self.wait(0.5)
        self.play(FadeOut(title), FadeOut(subtitle), run_time=0.5)
        # Total: 1.5+1.0+2.0+0.8+3.0+0.7+0.5+0.5 = 10.0s
        
        # Segment 2 (15.0s) - Triangle Introduction  
        A, B, C = ORIGIN + LEFT*2, RIGHT*2, RIGHT*2 + UP*3
        triangle = Polygon(A, B, C, color=WHITE)
        label_a = MathTex("A").next_to(A, DOWN+LEFT, buff=0.15)
        label_b = MathTex("B").next_to(B, DOWN+RIGHT, buff=0.15)
        label_c = MathTex("C").next_to(C, UP, buff=0.15)
        
        self.play(Create(triangle), run_time=2.0)
        self.play(Write(label_a), Write(label_b), Write(label_c), run_time=1.5)
        self.wait(3.0)
        self.play(Indicate(triangle), color=YELLOW, run_time=1.0)
